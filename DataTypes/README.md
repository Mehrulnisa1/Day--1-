# Integers

age = 21
marks = 85
year = 2026

print("Age:", age)
print("Marks:", marks)
print("Year:", year)

# Arithmetic operations
a = 10
b = 3

print("Addition:", a + b)
print("Subtraction:", a - b)
print("Multiplication:", a * b)
print("Division:", a / b)
print("Floor Division:", a // b)
print("Modulus:", a % b)
print("Power:", a ** b)

# Check data type
print(type(age))
